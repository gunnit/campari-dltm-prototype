  const [phase, setPhase] = React.useState('capture'); // 'capture' | 'scanning' | 'done'

  const detected = [
    { name: "Mojito", price: "9" },
    { name: "Gin Tonic", price: "8" },
    { name: "Moscow Mule", price: "9" },
    { name: "Margarita", price: "9" },
    { name: "Old Fashioned", price: "11" },
    { name: "Manhattan", price: "10" },
    { name: "Cosmopolitan", price: "9" },
    { name: "Whisky Sour", price: "10" },
    { name: "Espresso Martini", price: "10" },
    { name: "Caipirinha", price: "9" },
    { name: "Bloody Mary", price: "9" },
    { name: "Negroni Sbagliato", price: "10", note: "★ specialità" },
  ];

  React.useEffect(() => {

  const Brackets = () => (
    <>
      <div style={{ position: 'absolute', top: 14, left: 14, width: 20, height: 20, borderTop: '2px solid rgba(244,234,216,0.85)', borderLeft: '2px solid rgba(244,234,216,0.85)' }} />

    </>
  );

  // The photographed paper menu (placeholder for the locale's real photo).
  const PaperPhoto = ({ scanning }) => (
    <div style={{ position: 'absolute', inset: 0, display: 'flex', alignItems: 'center', justifyContent: 'center', background: 'radial-gradient(circle at 50% 32%, #3a2517, #160d07)', overflow: 'hidden' }}>

  // RIGHT — digitized result, per phase

'rgba(34,150,90,0.1)' : 'rgba(193,39,45,0.08)', border: `1px solid ${phase === 'done' ? 'rgba(34,150,90,0.25)' : 'rgba(193,39,45,0.2)'}`, borderRadius: 20, fontSize: 11, color: phase === 'done' ?           </div>
        </div>

          <div style={{ display: 'flex', alignItems: 'center', gap: 10, marginBottom: 10 }}>
            <div style={{ fontSize: 10, letterSpacing: 2, color: '#c1272d', textTransform: 'uppercase', fontWeight: 700 }}>Digitalizzato</div>

        {phase === 'done' ? (
          <>
            <button onClick={() => setPhase('capture')} style={{ padding: '10px 16px', background: 'transparent', border: '1.5px solid rgba(58,31,16,0.25)', borderRadius: 22, color: '#3a1f10', fontSize: 13, cursor: 'pointer', fontFamily: 'inherit' }}>↺ Riscansiona</button>

            </div>
          </>
        ) : (
          <div style={{ marginLeft: 'auto', fontFamily: '"Caveat", cursive', fontSize: 17, color: '#8a4a1a' }}>
            {phase === 'capture' ? 'acquisisci il menu per continuare →' : 'attendi la digitalizzazione →'}
